cmake_minimum_required(VERSION 3.13)
project(ESPressioObservableTests LANGUAGES CXX)

option(ESPRESSIO_ENABLE_COVERAGE "Enable source coverage instrumentation" OFF)
option(ESPRESSIO_ENABLE_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
set(ESPRESSIO_SYSTEM_DIR "${CMAKE_CURRENT_LIST_DIR}/../../ESPressio-System/src" CACHE PATH "Path to ESPressio System src")
if(NOT EXISTS "${ESPRESSIO_SYSTEM_DIR}/ESPressio_Memory.hpp")
    message(FATAL_ERROR "ESPressio System headers not found at ${ESPRESSIO_SYSTEM_DIR}")
endif()

set(OBSERVABLE_INCLUDES ../src ${ESPRESSIO_SYSTEM_DIR})
add_executable(espressio_observable_tests test_observable.cpp)
target_include_directories(espressio_observable_tests PRIVATE ${OBSERVABLE_INCLUDES})
target_compile_features(espressio_observable_tests PRIVATE cxx_std_17)

add_executable(espressio_observable_no_rtti_tests test_observable_no_rtti.cpp)
target_include_directories(espressio_observable_no_rtti_tests PRIVATE ${OBSERVABLE_INCLUDES})
target_compile_features(espressio_observable_no_rtti_tests PRIVATE cxx_std_17)

add_executable(espressio_observable_callback_boundary_tests test_threadsafe_callback_boundaries.cpp)
target_include_directories(espressio_observable_callback_boundary_tests PRIVATE ${OBSERVABLE_INCLUDES})
target_compile_features(espressio_observable_callback_boundary_tests PRIVATE cxx_std_17)

add_executable(espressio_observable_presence_tests test_observer_presence.cpp)
target_include_directories(espressio_observable_presence_tests PRIVATE ${OBSERVABLE_INCLUDES})
target_compile_features(espressio_observable_presence_tests PRIVATE cxx_std_17)

if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(espressio_observable_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_compile_options(espressio_observable_no_rtti_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-rtti)
    target_compile_options(espressio_observable_callback_boundary_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_compile_options(espressio_observable_presence_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
elseif(MSVC)
    target_compile_options(espressio_observable_tests PRIVATE /W4 /WX)
    target_compile_options(espressio_observable_no_rtti_tests PRIVATE /W4 /WX /GR-)
    target_compile_options(espressio_observable_callback_boundary_tests PRIVATE /W4 /WX)
    target_compile_options(espressio_observable_presence_tests PRIVATE /W4 /WX)
endif()

find_package(Threads REQUIRED)
target_link_libraries(espressio_observable_tests PRIVATE Threads::Threads)
target_link_libraries(espressio_observable_no_rtti_tests PRIVATE Threads::Threads)
target_link_libraries(espressio_observable_callback_boundary_tests PRIVATE Threads::Threads)
target_link_libraries(espressio_observable_presence_tests PRIVATE Threads::Threads)

if(ESPRESSIO_ENABLE_COVERAGE)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(espressio_observable_tests PRIVATE -O0 -g --coverage)
        target_link_options(espressio_observable_tests PRIVATE --coverage)
        target_compile_options(espressio_observable_callback_boundary_tests PRIVATE -O0 -g --coverage)
        target_link_options(espressio_observable_callback_boundary_tests PRIVATE --coverage)
        target_compile_options(espressio_observable_presence_tests PRIVATE -O0 -g --coverage)
        target_link_options(espressio_observable_presence_tests PRIVATE --coverage)
    else()
        message(FATAL_ERROR "Coverage is supported only with GCC or Clang")
    endif()
endif()

if(ESPRESSIO_ENABLE_SANITIZERS)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(espressio_observable_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(espressio_observable_tests PRIVATE -fsanitize=address,undefined)
        target_compile_options(espressio_observable_callback_boundary_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(espressio_observable_callback_boundary_tests PRIVATE -fsanitize=address,undefined)
        target_compile_options(espressio_observable_presence_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(espressio_observable_presence_tests PRIVATE -fsanitize=address,undefined)
    else()
        message(FATAL_ERROR "Sanitizers are supported only with GCC or Clang")
    endif()
endif()

enable_testing()
add_test(NAME espressio_observable_tests COMMAND espressio_observable_tests)
add_test(NAME espressio_observable_no_rtti_tests COMMAND espressio_observable_no_rtti_tests)
add_test(NAME espressio_observable_callback_boundary_tests COMMAND espressio_observable_callback_boundary_tests)
add_test(NAME espressio_observable_presence_tests COMMAND espressio_observable_presence_tests)
